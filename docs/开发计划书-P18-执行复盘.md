### 1.2 新增 `touchScanIn` / `touchScan`

放在 `computeGears` 之后、`renderGauge` 之前，独立一段：

```js
/* ===== P18 执行复盘 =====

   累计扫描次数：只给 test b2 断言「收起时一次都不扫」用，不参与业务 */
var RV_SCANS = 0;

/* 触价扫描：某个价位在日线里被触及过多少根、最近一次是哪根、区间最低多少。

   必须用 scaled("1d")：档位价的高点 H 来自 computeHEff()，它读的也是 scaled，
   两边同坐标系；用原始 S.kl["1d"] 会差一个 S.scale 倍，判出来的"触及"是错的。
   ——这一条与 P19-1 相反（那边只用相对关系，所以刻意用原始序列）。 */
function touchScanIn(bars, px){
  RV_SCANS++;
  if(!bars || bars.length < 30 || !(px > 0)) return null;
  var hit = 0, lastT = 0, low = Infinity, i, l;
  for(i = 0; i < bars.length; i++){
    l = Number(bars[i].l);
    if(!(l > 0)) continue;
    if(l < low) low = l;
    if(l <= px){ hit++; if(bars[i].t > lastT) lastT = bars[i].t; }
  }
  return {
    hit: hit, lastT: lastT || null, low: isFinite(low) ? low : null,
    from: bars[0].t, to: bars[bars.length - 1].t, n: bars.length
  };
}
function touchScan(px){ return touchScanIn(scaled("1d"), px); }
```

`bars.length < 30` 与 `fetchKlines` 的拒收阈值（`:1369`）一致，避免半截数据出结论。

### 1.3 新增 `rvRows()`

```js
/* 逐档复盘行。只读 S.gears 与标记，不写任何数据。

   触价扫描只对「待补且从没买过」的档跑：已执行 / 已周转的档不需要知道到没到价，
   省掉一半扫描。bars 只取一次，不要在循环里反复 scaled()。
   返回 { rows, win }：win 是日线区间（起止 + 根数），页脚要用它标统计范围。 */
function rvRows(){
  var gs = S.gears || [];
  var bars = scaled("1d");
  var win = (bars && bars.length >= 30)
    ? { from:bars[0].t, to:bars[bars.length - 1].t, n:bars.length } : null;
  var rows = [];
  gs.forEach(function(g){
    if(g.warn) return;
    var gross = Number(g.grossG) || 0;
    var sold  = Number(g.soldG) || 0;
    var net   = Number(g.boughtG) || 0;
    var plan  = Number(g.plan) || 0;
    var price = Number(g.price) || 0;
    var pending = Math.max(0, plan - net);
    var est = !!g.costEst;
    var avgOz = Number(g.avgOz) > 0 ? Number(g.avgOz) : null;
    /* 估算档的均价恒等于档位价，偏差必为 0 → 一律不给偏差，也不进加权 */
    var dev = (avgOz !== null && price > 0 && !est) ? (avgOz / price - 1) * 100 : null;

    var state, touch = null;
    if(!(plan > 0.05))        state = "none";
    else if(pending <= 0.05)  state = "done";
    else if(gross > 0.05)     state = "turned";
    else {
      touch = touchScanIn(bars, price);
      state = (touch && touch.hit > 0) ? "missed" : "far";
    }
    rows.push({
      idx:g.idx, name:g.name, price:price, plan:plan,
      gross:gross, sold:sold, net:net, pending:pending, over:Math.max(0, net - plan),
      avgOz:avgOz, est:est, dev:dev, state:state, touch:touch,
      realized:Number(g.realized) || 0, turns:Number(g.turns) || 0
    });
  });
  return { rows:rows, win:win };
}
```

**判定顺序不能换**：`done` 在 `turned` 前面。先判达标，再看是否买过 —— 「买过又卖了但仍达标」的档必须归 `done`；「买过又卖光了」的档因为 `pending > 0` 才落到 `turned`，这是预期。

### 1.4 新增 `rvSum()`

> 不复用 `planSummary()` 的 `grossBought / soldTotal / realizedTotal`：P20 批四已把这三个字段当死代码删除。这里的四个数从 `rows` 自己累加，口径相同（同源于 `gearMarks`），只是不共享变量。

```js
/* 合计与对账。加权只用「有真实成交价」的档：估算档的均价等于档位价，
   混进去会把真实偏差稀释成 0（构造 1 档 +5% + 5 档估算 → 会算成 +0.83%）。 */
function rvSum(rows){
  var gross = 0, sold = 0, net = 0, realized = 0, turns = 0;
  var wg = 0, wAmt = 0, wPlan = 0, graded = 0;
  rows.forEach(function(r){
    gross += r.gross; sold += r.sold; net += r.net;
    realized += r.realized; turns += r.turns;
    if(r.dev !== null && r.gross > 0.05){
      wg += r.gross; wAmt += r.gross * r.avgOz; wPlan += r.gross * r.price;
      graded++;
    }
  });
  var pAvg = wg > 0 ? wPlan / wg : null;
  var rAvg = wg > 0 ? wAmt / wg : null;
  return {
    gross:gross, sold:sold, net:net, realized:realized, turns:turns,
    pAvg:pAvg, rAvg:rAvg, graded:graded,
    devAll:(pAvg > 0 && rAvg !== null) ? (rAvg / pAvg - 1) * 100 : null,
    /* 每克拉低：已实现差价摊到现有持仓上，相当于每克成本降了多少（再折 $/oz） */
    perOz: net > 0.05 ? realized / net * OZ_PER_G : null,
    book: holdStats().bought
  };
}
```

### 2.3 单档三行块

```js
/* 状态文案与色号。missed 是唯一需要用户行动的，其余都是常态，不给压力色。
   橙点用 ● 字符（不是 emoji），全内联无外部依赖。 */
function rvStateText(r){
  if(r.state === "none")   return { t:"—", c:"dim" };
  if(r.state === "done")   return { t:"已执行" + (r.over > 0.05 ? " · 超 " + gnum(r.over) + "g" : ""), c:"ok" };
  if(r.state === "turned") return { t:"已周转", c:"dim" };
  if(r.state === "missed") return { t:"● 待补 · 到过价", c:"warn" };
  return { t:"待补 · 未到价", c:"dim" };
}

/* 偏差：买便宜绿、买贵橙、打平灰。不用红 —— 红在本页专指涨/亏损，
   0.3% 的溢价被标红会读成"亏了"。 */
function rvDevTag(r){
  if(r.dev === null) return '<b class="num dim">—</b>';
  var cls = Math.abs(r.dev) <= 0.3 ? "dim" : (r.dev < 0 ? "good" : "warn");
  var s = (r.dev < 0 ? "−" : "+") + num(Math.abs(r.dev), 2) + "%";
  return '<b class="num ' + cls + '">' + s + '</b>';
}

function rvBlock(r){
  var st = rvStateText(r);
  var r2;
  if(r.avgOz !== null){
    r2 = '<span>计划</span><b class="num">' + num(r.price, 1) + '</b>' +
         '<span>→</span><b class="num">' + num(r.avgOz, 1) + '</b>' +
         (r.est ? '<i class="rv-est">估</i>' : "") + rvDevTag(r);
  } else {
    r2 = '<span>计划</span><b class="num">' + num(r.price, 1) + '</b><span class="dim">· 未成交</span>';
  }
  return '<div class="rv-g">' +
    '<div class="rv-r1"><span class="rv-n">' + r.name + '</span><span class="' + st.c + '">' + st.t + '</span></div>' +
    '<div class="rv-r2">' + r2 + '</div>' +
    '<div class="rv-r3 dim">净持 <b class="num">' + gnum(r.net) + '</b> / <b class="num">' + gnum(r.plan) + '</b> g</div>' +
  '</div>';
}

function rvBlockList(rows){
  return '<div class="rv-list">' + rows.map(rvBlock).join("") + '</div>';
}
```

## 六、交付前自查（铁律 9 + 10）

1. **调用链无环**：`refreshAll → renderPos / renderReview`；`renderReview → rvRows → touchScanIn → scaled`；`renderReview → rvSum → holdStats`。**`renderReview` 不调用任何其他渲染函数**；`toggleReview → renderReview` 是单向的（`renderReview` 不回调 `toggleReview`）
2. **初始化链**：`RV_OPEN` 在脚本顶部初始化为 `false`，`renderReview` 首调时 `#posReview` 已存在（DOM 在 `<script>` 之前）
3. **DOM 存在性**：`$("posReview")` 有 null 保护；`box.querySelector('[data-rv="h"]')` 结果也判空
4. **空数据不崩**：无标记 → 照常出明细 + 表尾引导；计划克重全 0 → 走 `rv-empty`，不进 `rvBlockList`
5. **除零**：`plan = 0`、`gross = 0`、`net = 0`、`pAvg = 0` 四条路径都返回 `null` 并渲染 `—`
6. **事件绑定时机**：每次 `innerHTML` 赋值之后重新绑 `onclick`（收起 / 展开 / 空态三条分支都要绑，否则切一次就点不动）
7. **变量作用域**：`RV_OPEN` 是模块级单例，不在函数内重声明；`touchScanIn` 的 `low` 用 `Infinity` 初值，一根有效 low 都没有时返回 `null` 而不是 `Infinity`
8. **模块数**：本期 2 个子模块（逐档表 / 合计对账），合规
