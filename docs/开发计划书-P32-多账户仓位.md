## 1 先纠一处 PRD 的口径错误

PRD 验收第 13 条写「账户卡片三项（克重 / 成本 / 浮盈）相加 = 全局值」。**这条不成立，必须改。**

原因在于「卖出不改变均价」这个既有口径：`hs.avg` 是**全量买入加权均价**，卖 5 克不重新分摊成本。于是：

```
招行 买 10g@900          工行 买 10g@1000
全局  avg=950  grams=20  cost=19,000
账户  9,000 + 10,000 = 19,000   ← 相等

招行卖 5g@1100 之后
全局  avg=950  grams=15  cost=14,250
账户  招行 5×900=4,500 + 工行 10,000 = 14,500   ← 差 250
```

差额 = `Σ grams_i × (avg_global − avg_i)`。卖出时按全局均价结转成本、账户按各自均价结转，两个口径天然分叉。浮盈同理。

**裁决：**

| 量 | 能不能相加 |
|---|---|
| 剩余克重 | ✅ 相等（这是唯一的硬对账） |
| 买入总额 / 卖出总额 | ✅ 相等 |
| 剩余成本 `grams × avg` | ❌ 不等，**不做相加断言** |
| 浮动盈亏 | ❌ 不等，**卡片头直接用全局 `holdStats()` 的值，不由账户行累加** |

要让它们相等，只能把成本核算改成「账户内加权平均 / FIFO 出账」，那是另一套模型，改动面超过整个 P32。一期不做，理由写进卡片脚注：

> 账户行是各账户自己的口径；卡片顶部三个数取自全账户汇总，不由下面几行累加。

验收第 13 条改为：`Σ 账户克重 = 全局克重（±0.001）`、`Σ 账户买入额 = 全局买入额（±0.5 元）`，并新增一条反向断言：**账户行浮盈之和与全局浮盈不要求相等**（防止后来的人"顺手加一下"）。

### 2.2 `holdStats(acct)`

| 项 | 现在 | 改后 |
|---|---|---|
| 入参 | 无 | `acct`：账户 id，缺省/`null` = 全局 |
| 返回结构 | 不变 | 不变（字段一个不加，避免波及 18 个调用点） |

```js
function holdStats(acct){
  var all = !acct;
  var buys  = (S.cfg.buys ||[]).slice();
  var sells = (S.cfg.sells||[]).slice();
  if(!all){
    buys  = buys.filter(function(b){ return acctOf(b.acct) === acct; });
    sells = sells.filter(function(x){ return acctOf(x.acct) === acct; });
  }
  var bg=0, amt=0, sg=0, sAmt=0;
  buys.forEach(function(b){
    var gg = Number(b.gram)||0, pp = Number(b.priceCny)||0;
    bg += gg; amt += gg*pp;
  });
  /* 老兼容只在全局时走：筛选空账户不能拿到这份全局汇总 */
  if(all && !buys.length && Number(S.cfg.GRAMS)>0 && Number(S.cfg.COST)>0){
    bg = Number(S.cfg.GRAMS); amt = bg*Number(S.cfg.COST);
  }
  sells.forEach(function(s){
    var gg = Number(s.gram)||0, pp = Number(s.priceCny)||0;
    sg += gg; sAmt += gg*pp;
  });
  var avg = bg ? amt/bg : 0;
  return { buys:buys, sells:sells, bought:bg, sold:sg,
           grams: Math.max(0, bg-sg), amount:amt, avg:avg,
           soldAmt:sAmt, realized: avg ? sAmt - sg*avg : 0 };
}
```

> 老兼容那段的 `all &&` 是必须的。漏了它，筛一个空账户会显示"历史汇总 100 克"，是最容易被漏测的坑。

### 2.3 三个访问器（新增，UTIL 层）

```js
function acctList(){                       /* 至少保证 1 个，坏数据不抛错 */
  var a = S.cfg && S.cfg.accounts;
  if(!Array.isArray(a) || !a.length) return [];
  return a;
}
function acctOf(id){                       /* 失效 id 回落第一个，绝不丢记录 */
  var a = acctList();
  if(!a.length) return null;
  for(var i=0;i<a.length;i++) if(a[i].id === id) return id;
  return a[0].id;
}
function acctName(id){
  var a = acctList(), k = acctOf(id);
  for(var i=0;i<a.length;i++) if(a[i].id === k) return a[i].name;
  return "默认账户";
}
```

### 2.4 `acctStats()`（新增）

单遍聚合，不调 N 次 `holdStats`（账户多时是 O(n·m)）。

```js
/* 返回 [{id,name,note,color,buyGram,gram,amt,avg,costLeft,pnl,recs,share}]。
   pnl 在现价/汇率为空时为 null（不是 0）。金额单位元，克重单位 g。 */
function acctStats(){
  var list = acctList(), map = {}, order = [];
  list.forEach(function(a){
    map[a.id] = { id:a.id, name:a.name, note:a.note||"", color:a.color||"#D4A24C",
                  buyGram:0, sold:0, gram:0, amt:0, soldAmt:0, recs:0 };
    order.push(a.id);
  });
  function slot(id){ var k = acctOf(id); return k ? map[k] : null; }

  (S.cfg.buys||[]).forEach(function(b){
    var s = slot(b.acct); if(!s) return;
    var g = Number(b.gram)||0, p = Number(b.priceCny)||0;
    s.buyGram += g; s.amt += g*p; s.recs++;
  });
  (S.cfg.sells||[]).forEach(function(x){
    var s = slot(x.acct); if(!s) return;
    var g = Number(x.gram)||0, p = Number(x.priceCny)||0;
    s.sold += g; s.soldAmt += g*p; s.recs++;
  });

  var nowCny = (S.fx && S.price) ? S.price*S.fx/OZ : null;
  var totalAll = holdStats().grams;              /* 分母用全局，不是循环累加 */
  return order.map(function(id){
    var r = map[id];
    r.gram = Math.max(0, r.buyGram - r.sold);
    /* 均价分母必须是买入总克重 buyGram，不是卖出后的余额 gram ——
       用 gram 会让"卖过一半"的账户均价虚高。 */
    r.avg      = r.buyGram > 0 ? r.amt / r.buyGram : 0;
    r.costLeft = r.gram * r.avg;
    r.pnl      = nowCny !== null ? r.gram * (nowCny - r.avg) : null;
    r.share    = totalAll > 0 ? r.gram / totalAll : 0;
    return r;
  });
}
```

**两个容易写错的点，都要留注释**：

1. `buyGram` 与 `gram` 必须分开存。均价分母是买入总克重，余额是扣掉卖出的克重，混用会让卖过一半的账户均价虚高。
2. `share` 的分母用全局 `holdStats().grams`，不能用循环里累加的那个 —— 累加值在有账户克重为 0 时偏小，占比会超过 100%。

> 三个分支都要在 `innerHTML` 之后补一次事件绑定。抽一个 `bindAcctBar(box)`，在这三处和正常分支末尾各调一次 —— 比复制三遍 onclick 稳。

- 校验：重名（同 id 除外）、空名、超长
- 删除：
  - 无记录 → `confirm` 二次确认 → 移出 `accounts` + 写 `delAccts`
  - 有记录 → 弹「转移到…」选择器（`select` 列出其余账户）→ 批量改 `acct` → 再删
  - 只剩 1 个 → `toast("至少保留一个账户")`
- 「管理」入口只放一处：账户卡片的标题右侧（`账户 3 · 管理`）。筛选条里不放，避免同屏两个同样的按钮

转移实现（只改归属，不产生划转记录，不改金额）：

```js
function moveAcct(from, to){
  var n = 0;
  (S.cfg.buys ||[]).forEach(function(b){ if(acctOf(b.acct)===from){ b.acct = to; n++; } });
  (S.cfg.sells||[]).forEach(function(x){ if(acctOf(x.acct)===from){ x.acct = to; n++; } });
  if(S.cfg.acctLast === from) S.cfg.acctLast = to;
  return n;      /* 返回转移笔数，用于 toast */
}
```
